\section{Основные определения и свойства}

\subsection{Введение}

\begin{definition}
	Матрица размера $m \times n$ -- это прямоугольная таблица, состоящая
	из $m$ строк и $n$ столбцов. Множество матриц размера $m \times n$
	над полем $F$ обозначается как $\mat_{m \times n}(F)$.
\end{definition}
\begin{definition}
	Матрицы $A,B\in \mat_{m \times n}$ называются равными, если:
	$$\forall 1 \leqslant i \leqslant m, 1 \leqslant j \leqslant
	n\ a_{ij} = b_{ij}$$
\end{definition}
\begin{definition}
	Пространством $\R^n$ называется:
	$$\R^n \coloneqq \{(x_1, x_2, \ldots, x_n) \mid x_i \in \R\}$$
\end{definition}
\begin{definition}
	Матрица $A\in \mat_{n \times n}$ (или $A \in M_n$) называется
	квадратной порядка $n$.
\end{definition}
\begin{definition}
	Матрица $A\in M_n$ называется диагональной, если:
	$$\forall i, j, i \neq j\ a_{ij}=0$$
	Такие матрицы обозначаются $\diag(a_1, \ldots, a_n)$, где $a_k$ --
	$k$-й элемент главной диагонали.
\end{definition}
\begin{definition}
	Матрица $E_n=\diag(1,\ldots,1)$ называется единичной матрицей порядка $n$.
\end{definition}
\begin{definition}
	Пусть $A\in \mat_{m \times n}$. Транспонированной матрицей $B=A^T\in
	\mat_{n \times m}$ называется матрица, для которой:
	$$\forall i, j\ b_{ij}=a_{ji}$$
\end{definition}

\subsection{Сложение матриц и умножение на скаляр}

\begin{definition}
	Пусть $A,B \in \mat_{m \times n}$. Суммой матриц $A$ и $B$ называется:
	$$A + B \coloneqq C \in \mat_{m \times n}\colon c_{ij} = a_{ij} +
	b_{ij}\ \forall 1 \leqslant i \leqslant m, 1 \leqslant j \leqslant n$$
\end{definition}
\begin{definition}
	Пусть $A\in \mat_{m \times n}, \lambda \in \R$. Умножением матрицы
	$A$ на скаляр $\lambda$ называется:
	$$\lambda A \coloneqq B \in \mat_{m \times n}\colon b_{ij} = \lambda
	a_{ij}\ \forall 1 \leqslant i \leqslant m, 1 \leqslant j \leqslant n$$
\end{definition}
\begin{statement}[Свойства сложения матриц и умножения на скаляр]
	Пусть $A,B,C \in \mat_{m \times n}, \lambda, \mu \in \R$. Тогда:
	\begin{center}
		\begin{minipage}{0.75\textwidth}
			\begin{enumerate}[itemsep=0pt]
				\item $A + B = B + A$ (коммутативность сложения);
				\item $(A + B) + C = A + (B + C)$ (ассоциативность сложения);
				\item $A + 0 = A$ (нейтральный элемент по сложению);
				\item $1\cdot A = A$ (нейтральный элемент по умножению на скаляр);
				\item $A + (-A) = 0$ (противоположный элемент по сложению);
				\item $\lambda(A + B) = \lambda A + \lambda B$ (дистрибутивность
					относительно сложения);
				\item $(\lambda + \mu)A = \lambda A + \mu A$ (дистрибутивность
					относительно скаляров);
				\item $\lambda(\mu A) = (\lambda\mu)A$ (ассоциативность умножения
					на скаляр).
			\end{enumerate}
		\end{minipage}
	\end{center}
\end{statement}

\subsection{Умножение матриц}

\begin{definition}
	Пусть $A \in \mat_{m \times n}$, $B \in \mat_{n \times p}$,
	умножением матриц $A$ и $B$ называется:
	$$AB \coloneqq C \in \mat_{m \times p} \colon
	c_{ij}=A_{(i)}B^{(j)}=\sum_{k=1}^{n}a_{ik}\cdot b_{kj}$$
\end{definition}
\begin{statement}[Левая дистрибутивность умножения]
	$$A(B+C) = AB + AC;\  A \in \mat_{m \times n}, B, C \in \mat_{n
	\times p}$$
	\begin{proof}
		Пусть $X=A(B+C), Y=AB + AC$. Тогда:
		$$x_{ij}=\sum_{k=1}^na_{ik}(b_{kj}+c_{kj})=\sum_{k=1}^{n}a_{ik}b_{kj}+\sum_{k=1}^{n}a_{ik}c_{kj}=y_{ij}$$
	\end{proof}
\end{statement}
\begin{statement}[Правая дистрибутивность умножения]
	$$(A+B)C = AC + BC;\  A, B \in \mat_{m \times n}, C \in \mat_{n
	\times p}$$
\end{statement}
\begin{statement}
	$$\lambda(AB)=(\lambda A)B=A(\lambda B);\ A \in \mat_{m \times n}, B
	\in \mat_{n \times p}, \lambda \in \R$$
\end{statement}
\begin{statement}[Ассоциотивность умножения]
	$$(AB)C=A(BC);\ A \in \mat_{m \times n}, B \in \mat_{n \times p}, C
	\in \mat_{p \times q}$$
	\begin{proof}
		Пусть $X=(AB)C, Y=A(BC), U=AB, V=BC$. Тогда:
		$$x_{ij}=\sum_{k=1}^p u_{ik}c_{kj} =
		\sum_{k=1}^{p}\left(\sum_{l=1}^{n}a_{il}b_{lk}\right)c_{kj}=\sum_{l=1}^{n}\left(a_{il}\sum_{k=1}^{p}b_{lk}c_{kj}\right)=\sum_{l=1}^{n}a_{il}v_{lj}=y_{ij}$$
	\end{proof}
\end{statement}
\begin{statement}
	$$(AB)^T=B^TA^T;\ A \in \mat_{m \times n}, B \in \mat_{n \times p}$$
	\begin{proof}
		Пусть $X=(AB)^T, Y=B^TA^T$. Тогда:
		$$x_{ij}=(AB)_{ji}=A_{(j)}B^{(i)}=(B^T)_{(i)}(A^T)^{(j)}=y_{ij}$$
	\end{proof}
\end{statement}

\subsection{След матрицы}

\begin{definition}
	Следом матрицы называется сумма элементов на её главной диагонали.
\end{definition}
\begin{statement}[Свойства следа]
	\hfill
	\begin{center}
		\begin{minipage}{0.35\textwidth}
			\begin{enumerate}[itemsep=0pt]
				\item $\tr(A+B)=\tr(A)+\tr(B)$;
				\item $\forall \lambda \in \R\ \tr(\lambda A)=\lambda\tr(A)$;
				\item $\tr(A^T)=\tr(A)$.
			\end{enumerate}
		\end{minipage}
	\end{center}
\end{statement}
\begin{statement}
	$$\forall A\in \mat_{m \times n}, B \in \mat_{n \times m}\ \tr(AB)=\tr(BA)$$
	\begin{proof}
		Пусть $X=AB, Y=BA$. Тогда:
		$$\tr(X)=\sum_{i=1}^{m}x_{ii}=\sum_{i=1}^{m}\left(\sum_{j=1}^{n}a_{ij}b_{ji}\right)=\sum_{j=1}^{n}y_{jj}=\tr(Y)$$
	\end{proof}
\end{statement}

\section{Системы линейных уравнений}

\begin{definition}
	СЛУ называется совместной, если у неё есть хотя бы одно решение.
\end{definition}
\begin{definition}
	Две СЛУ от одних и тех же неизвестных называются эквивалентными,
	если множество их решений совпадает.
\end{definition}

\subsection{Элементарные преобразования СЛУ}

\begin{definition}[Элементарные преобразование СЛУ]
	\hfill
	\begin{center}
		\begin{minipage}{0.7\textwidth}
			\begin{enumerate}[itemsep=0pt]
				\item К $i$-му уравнению прибавить $j$-е, домноженное на $\lambda
					\in R, i \neq j$;
				\item Поменять местами $i$-ю и $j$-ю строки, $i \neq j$;
				\item Умножить $i$-е уравнение на $\lambda \in \R \setminus \{0\}$.
			\end{enumerate}
		\end{minipage}
	\end{center}
\end{definition}
\begin{lemma}
	Элементарные преобразования СЛУ сохраняют множество её решений.
	\begin{proof}
		Заметим, что для всякого элементарного преобразования СЛУ
		существует обратное:
		\begin{center}
			\begin{tblr}{|c|c|}
				\hline
				Прямое преобразование & Обратное преобразование\\
				\hline
				Э$_1(i, j, \lambda)$ & Э$_1(i, j, -\lambda)$\\
				Э$_2(i, j)$ & Э$_2(i, j)$\\
				Э$_3(i, \lambda)$ & Э$_3(i, \frac{1}{\lambda})$\\
				\hline
			\end{tblr}
		\end{center}
		Значит всякое решение СЛУ сохраняется при её элементарном преобразовании.
	\end{proof}
\end{lemma}
\begin{definition}
	Ведущим элементом строки называется её первый ненулевой элемент.
\end{definition}
\begin{definition}
	Матрица называется ступенчатой (или имеет ступенчатый вид), если
	номера ведущих элементов её ненулевых строк строго возрастают, а все
	нулевые строки находятся в конце.
\end{definition}
\begin{definition}
	Матрица имеет улучшенный ступенчатый вид, если:
	\begin{center}
		\begin{minipage}{0.71\textwidth}
			\begin{enumerate}[itemsep=0pt]
				\item Она имеет ступенчатый вид;
				\item Ведущие элементы всех ненулевых строк равны 1;
				\item В одном столбце с ведущим элементом находятся только нули.
			\end{enumerate}
		\end{minipage}
	\end{center}
\end{definition}
\begin{theorem}
	Всякую матрицу элементарными преобразованиями можно привести к
	улучшенному ступенчатому виду.
\end{theorem}
\begin{definition}
	СЛУ называется однородной (ОСЛУ), если все её правые части равны нулю.
\end{definition}
\begin{statement}
	Всякая ОСЛУ имеет тривиальное решение и всякая ОСЛУ, у которой
	переменных больше, чем уравнений, имеет нетривиальное решение.
\end{statement}
\begin{definition}
	Пусть $A\in \mat_{m \times n}(\R), x\in \R^n, b\in \R^m$. Если СЛУ
	$Ax=b$ совместна, то её частое решение -- это какое-нибудь одно её решение.
\end{definition}
\begin{lemma}
	Пусть $A\in \mat_{m \times n}(\R), x\in \R^n, b\in \R^m$, СЛУ $Ax=b$
	совместна, $L\subseteq \R^n$ -- её множество решений,
	$x_0\in L$ -- её частное решение, $S\subseteq \R^n$ -- множество
	решений ОСЛУ. Тогда:
	$$L=x_0+S\text{, где }x_0+S\coloneqq \{x_0+v\mid v\in S\}$$
	\begin{proof}
		Так как $x_0\in L$, имеем $Ax_0=b$. Рассмотрим $u\in L$. Пусть
		$v=u-x_0$, то есть $u=x_0+v$, значит:
		$$Av=A(u-x_0)=Au-Ax_0=b-b=0\implies v\in S\implies u\in
		x_0+S\implies L\subseteq x_0+S$$
		Теперь докажем, что $x_0+S$ также подмножество $L$:
		$$v\in S\implies Av=0\implies A(x_0+v)=Ax_0+Av=b+0=b\implies
		x_0+v\in L\implies x_0+S\subseteq L$$
	\end{proof}
\end{lemma}
\begin{statement}
	Всякое элементарное преобразование строк матрицы $A\in \mat_{m\times
	n}$ реализуется при помощи умножения слева на подходящую
	элементарную матрицу $U_i$. При этом, чтобы получить такую
	элементарную матрицу $U_i$, нужно произвести
	элементарное преобразование $\text{Э}_i$ с единичной матрицей.\bigskip

	Аналогично утверждение верно для преобразования столбцов, но тогда
	умножать на элементарные матрицы надо справа.
\end{statement}
\begin{statement}
	Всякое элементарное преобразование столбцов матрицы $A\in \mat_{m\times
	n}$ реализуется при помощи умножения справа на подходящую
	элементарную матрицу. Причём элементарные матрицы для преобразования
	столбцов будут иметь вид, аналогичный транспонированным элементарным
	матрицам для преобразования строк.
\end{statement}

\subsection{Перестановки}

\begin{definition}
	Перестановкой (или подстановкой) множества $M=\{1, \ldots, n\}$
	называется биективное отображение $\sigma\colon M\to M$. Множество
	$S_n$ -- множества всех таких перестановок.
\end{definition}
\begin{definition}
	Пусть $\sigma \in S_n, i, j\in \{1, \ldots, n\}, i\neq j$. Говорят,
	что неупорядоченная пара $\{i, j\}$ образует инверсию в $\sigma$,
	если $i-j$ и $\sigma(i)-\sigma(j)$ имеют разный знак.
\end{definition}
\begin{definition}
	Знак перестановки $\sigma \in S_n$ -- это число:
	$$\sgn(\sigma)\coloneqq (-1)^{\text{число инверсий в }\sigma}$$
\end{definition}
\begin{definition}
	Перестановка $\sigma \in S_n$ называется чётной, если
	$\sgn(\sigma)=1$. Перестановка называется нечётной, если $\sgn(\sigma)=-1$.
\end{definition}
\begin{definition}
	Произведением (или композицией) двух перестановок $\sigma, \rho\in
	S_n$ называется перестановка $\sigma\cdot\rho$, действующая по правилу:
	$$\forall x\in \{1,2,\ldots,n\}\ (\sigma\cdot \rho)(x)=\sigma(\rho(x))$$
\end{definition}
\begin{statement}[Умножение перестановок ассоциативно]
	$$\forall \sigma, \rho, \pi\in
	S_n\ (\sigma\rho)\pi=\sigma(\rho\pi)$$
	\begin{proof}
		$$\forall \sigma, \rho, \pi\in
		S_n\ ((\sigma\rho)\pi)(x)=(\sigma\rho)(\pi(x))=\sigma(\rho(\pi(x))),
		(\sigma(\rho\pi))(x)=\sigma((\rho\pi)(x))=\sigma(\rho(\pi(x)))$$
	\end{proof}
\end{statement}
\begin{definition}
	Перестановка $\id\colon x\mapsto x, \id\in S_n$ называется тождественной.
\end{definition}
\begin{definition}
	Для всякой перестановки $\sigma\in S_n$ перестановка
	$\sigma^{-1}\colon \sigma(x)\mapsto x$ называется обратной к $\sigma$.
\end{definition}
\begin{definition}
	Транспозициями называются перестановки $\tau_{ij}, i\neq j$ вида:
	$$\tau_{ij}(i)=j, \tau_{ij}(j)=i, \forall k\neq i, j\ \tau_{ij}(k)=k$$
\end{definition}
\begin{definition}
	Перестановки вида $\tau_{i,i+1}$ называются элементарными транспозициями.
\end{definition}
\begin{statement}
	Всякая перестановка $\sigma\in S_n$ представима в виде произведения
	конечного числа элементарных транспозиций.
	\begin{proof}
		Применим сортировку «пузырьком« к $\sigma$, тогда
		$\sigma\tau_1\ldots\tau_k=\id$. Умножим равенство справа на
		$\tau_k\ldots\tau_1$, поскольку $\tau^2=\id$, получаем
		$\sigma=\tau_k\ldots\tau_1$.
	\end{proof}
\end{statement}
\begin{lemma}
	Число инверсий в $\sigma$ и $\sigma\tau$ отличается на 1.
\end{lemma}
\begin{consequence}
	Пусть $\sigma\in S_n$ и $\sigma=\tau_1\ldots\tau_k$, тогда
	$\sgn(\sigma)=(-1)^k$.
\end{consequence}
\begin{consequence}
	Для произвольной транспозиции $\tau_{ij}\in S_n$ $\sgn(\tau_{ij})=-1$.
	\begin{proof}
		Пусть $i<j$ (доказательство при $i>j$ аналогично), тогда:
		$$\tau_{ij}=\tau_{i,i+1}\cdot\tau_{i+1,i+2}\cdot\ldots\cdot\tau_{j-2,j-1}\cdot\tau_{j-1,j}\cdot\tau_{j-2,j-1}\cdot\ldots\cdot\tau_{i,i+1}$$
		Получаем произведение $2(j-i-1)+1$ элементарных транспозиций, то
		есть $\sgn(\tau_{ij})=-1$.
	\end{proof}
\end{consequence}
\begin{theorem}
	$$\forall \sigma,\rho\in S_n\ \sgn(\sigma\rho)=\sgn(\sigma)\cdot\sgn(\rho)$$
	\begin{proof}
		Пусть $\sigma=\tau_1\ldots\tau_k, \rho=\tau_1'\ldots\tau_m'$, тогда
		$\sgn(\sigma)=(-1)^k, \sgn(\rho)=(-1)^m$, при этом:
		$$\sigma\rho=\tau_1\ldots\tau_k\tau_1'\ldots\tau_m'\implies
		\sgn(\sigma\rho)=(-1)^{k+m}=\sgn(\sigma)\cdot\sgn(\rho)$$
	\end{proof}
\end{theorem}
\begin{consequence}
	$$\forall \sigma\in S_n\ \sgn(\sigma)=\sgn(\sigma^{-1})$$
\end{consequence}
\begin{statement}
	Пусть $\tau\in S_n$ -- некоторая транспозиция, тогда отображение
	$\sigma\mapsto\sigma\tau$ задаёт биекцию между множествами чётных и
	нечётных перестановок в $S_n$.
\end{statement}

\section{Определители}

\begin{definition}
	Пусть $A\in M_n$, определителем матрицы $A$ называется величина:
	$$\det A=|A|=\sum_{\sigma\in S_n}\sgn \sigma\cdot
	a_{1,\sigma(1)}a_{2,\sigma(2)}\ldots a_{n,\sigma(n)}$$
\end{definition}
\begin{statement}
	$$\det A = \det A^T, A\in M_n$$
	\begin{proof}
		Пусть $B=A^T$, тогда:
		\begin{multline*}
			\det B=\sum_{\sigma\in S_n}\sgn \sigma\cdot
			b_{1,\sigma(1)}b_{2,\sigma(2)}\ldots
			b_{n,\sigma(n)}=\sum_{\sigma\in S_n}\sgn \sigma\cdot
			a_{\sigma(1), 1}a_{\sigma(2),2}\ldots
			a_{\sigma(n),n}=\\
			=\sum_{\sigma\in S_n}\sgn \sigma^{-1}\cdot
			a_{1, \sigma^{-1}(1)}a_{2,\sigma^{-1}(2)}\ldots
			a_{n,\sigma^{-1}(n)}=\sum_{\rho\in S_n}\sgn \rho\cdot
			a_{1, \rho(1)}a_{2,\rho(2)}\ldots
			a_{n,\rho(n)}=\det A
		\end{multline*}
	\end{proof}
\end{statement}
\begin{statement}
	Если в $A$ есть нулевая строка или столбец, то $\det A=0$.
	\begin{proof}
		Пусть $A_{(i)}$ нулевая, тогда $\forall\sigma\in
		S_n\ a_{i,\sigma(i)}=0$, то есть каждое слагаемое равно нулю. Для
		столбцов верно, так как $\det A=\det A^T$.
	\end{proof}
\end{statement}
\begin{statement}
	Если в $A$ некоторую строку или столбец умножить на скаляр
	$\lambda$, то $\det A$ умножится на $\lambda$.
	\begin{proof}
		Пусть $A_{(i)}$ умножили на $\lambda$, тогда $\forall\sigma\in
		S_n\ a_{i,\sigma(i)}$ умножился на $\lambda$, то есть каждое
		слагаемое умножилось на $\lambda$. Для столбцов верно, так как
		$\det A=\det A^T$.
	\end{proof}
\end{statement}
\begin{consequence}
	$$\det(\lambda A)=\lambda^n\det A, A\in M_n$$
\end{consequence}
\begin{statement}
	Пусть $A_{(i)}=A_{(i)}'+A_{(i)}''$, тогда $\det A=\det A'+\det A''$,
	где $A'$ -- матрица $A$, в которой $i$-ую строку заменили на
	$A_{(i)}'$, $A''$ -- матрица $A$, в которой $i$-ую строку заменили
	на $A_{(i)}''$. Аналогичное утверждение верно для столбцов.
	\begin{proof}
		По условию $a_{ij}=a_{ij}'+a_{ij}''$. Тогда получаем, что:
		\begin{multline*}
			\det A=\sum_{\sigma\in S_n}\sgn \sigma\cdot
			a_{1,\sigma(1)}a_{2,\sigma(2)}\ldots(a_{i,\sigma(i)}'+a_{i,\sigma(i)}'')
			a_{n,\sigma(n)}=\\
			=\sum_{\sigma\in S_n}\sgn \sigma\cdot
			a_{1,\sigma(1)}a_{2,\sigma(2)}\ldots a_{i,\sigma(i)}'\ldots
			a_{n,\sigma(n)}+\sum_{\sigma\in S_n}\sgn \sigma\cdot
			a_{1,\sigma(1)}a_{2,\sigma(2)}\ldots a_{i,\sigma(i)}''\ldots
			a_{n,\sigma(n)}=\\
			=\det A' + \det A''
		\end{multline*}
		Для столбцов верно, так как $\det A=\det A^T$.
	\end{proof}
\end{statement}
\begin{statement}
	Если в $A$ есть две одинаковые строки или столбца, то $\det A=0$.
	\begin{proof}
		Пусть $A_{(i)}=A_{(j)}, i<j$, $\tau=\tau_{ij}\in S_n$ --
		соответствующая транспозиция. Достаточно доказать, что для каждой
		пары $\{\sigma, \sigma\tau\}$ (чётная и нечётная перестановка) два
		соответствующих слагаемых в сумме дают 0. Слагаемое для $\sigma$ имеет вид:
		$$\alpha=\sgn \sigma\cdot a_{1,\sigma(1)}a_{2,\sigma(2)}\ldots
		a_{n,\sigma(n)}$$
		Тогда применяя то, что $i$ и $j$ строки совпадают, слагаемое для
		$\sigma\tau$ имеет вид:
		$$\beta=\sgn (\sigma\tau)\cdot
		a_{1,(\sigma\tau)(1)}a_{2,(\sigma\tau)(2)}\ldots
		a_{n,(\sigma\tau)(n)}=-\sgn\sigma\cdot
		a_{1,\sigma(1)}a_{2,\sigma(2)}\ldots a_{i, \sigma(j)}\ldots
		a_{j,\sigma(i)}\ldots a_{n,\sigma(n)}=-\alpha$$
		Для столбцов верно, так как $\det A=\det A^T$.
	\end{proof}
\end{statement}
\begin{statement}
	Если к одной строке (или столбцу) матрицы $A$ прибавить другую
	строку (или столбец), умноженную на скаляр, то $\det A$ не изменится.
	\begin{proof}
		Рассмотрим такую $A'$, что $A'_{(i)}=A_{(i)}+A_\lambda{(j)}, i\neq
		j$, и такую $A''$, что $A''_{(i)}=\lambda A_{(j)}$ Тогда $\det
		A'=\det A + \det A''=\det A$, так как в $A''$ две строки совпадают,
		то есть $\det A''=0$. Для столбцов верно, так как $\det A=\det A^T$.
	\end{proof}
\end{statement}
\begin{statement}
	Если в $A$ поменять местами две строки или столбца, то $\det A$
	поменяет знак, то есть умножится на $-1$.
	\begin{proof}
		Пусть $A'$ -- матрица, в которой поменяли местами $i$-ую и $j$-ую
		строки. Тогда её можно получить прибавляя и вычитая строки $A$,
		после чего домножить одну из строк на $-1$, откуда $\det A'=-\det
		A$. Для столбцов верно, так как $\det A=\det A^T$.
	\end{proof}
\end{statement}
\begin{definition}
	Матрица $A$ называется верхнетреугольной (нижнетреугольной), если
	$\forall i>j\ a_{ij}=0$ ($\forall i<j\ a_{ij}=0$).
\end{definition}
\begin{statement}
	Если $A$ верхнетреугольна (нижнетреугольна), то $\det A=a_{11}\cdot
	a_{22}\cdot\ldots\cdot a_{nn}$.
	\begin{proof}
		Рассмотрим в сумме определения определителя верхнетреугольной $A$
		слагаемые, отличные от нуля. Пусть для $\sigma\in S_n$
		соответствующее слагаемое $\alpha$ не равно нулю, тогда:
		$$\sgn \sigma\cdot a_{1,\sigma(1)}a_{2,\sigma(2)}\ldots a_{n,\sigma(n)}\neq0$$
		$a_{n,\sigma(n)}\neq0\implies \sigma(n)=n,
		a_{n-1,\sigma(n-1)}\neq0\implies \sigma(n-1)\in\{n-1, n\}$, но
		$\sigma$ -- перестановка, то есть инъективна, значит
		$\sigma(n-1)=n-1$. Аналогично получаем, что $\sigma=\id$, то есть
		$\alpha\neq0\implies \sigma=\id$, значит единственное ненулевое
		слагаемое равно произведению элементов на главной диагонали. Для
		нижнетреугольной верно, так как $\det A=\det A^T$.
	\end{proof}
\end{statement}
\begin{statement}[Определитель с углом нулей]
	Пусть матрица $A\in M_n$ имеет блочный вид:
	$$A=
	\begin{+pmatrix}
		P & Q\\
		0 & R
	\end{+pmatrix}\text{ или }A=
	\begin{+pmatrix}
		P & 0\\
		Q & R
	\end{+pmatrix}$$
	Причём $P$ и $R$ квадратные, тогда $\det A=\det P\cdot \det R$.
	\begin{proof}
		Пусть $A$ -- марица с левым нижним нулевым углом. Если матрицы $P,
		R$ верхнетреугольны, то $A$ верхнетреугольна, тогда утверждение
		верно. Выполняя элементарные преобразования строк в блоке $(P\;Q)$,
		приведём его к виду $(P'\;Q')$, где $P'$ -- верхнетреугольная, при
		этом $\det A$ и $\det P$ умножаются на один и тот же ненулевой
		скаляр $\alpha$. Аналогично приведём блок $(0\;R)$ к виду
		$(0\;R')$, где $R'$ -- верхнетреугольная, при этом $\det A$ и $\det
		R$ умножаются на один и тот же ненулевой скаляр $\beta$. Тогда рассмотрим:
		$$A'=
		\begin{+pmatrix}
			P' & Q'\\
			0 & R'
		\end{+pmatrix}$$
		$$\det A'=\det P'\cdot \det R'=\alpha\det P\cdot \beta \det
		R=\alpha\beta\det A$$
		Сокращая на $\alpha\beta\neq0$, получаем искомое утверждение. Для
		второго случая аналогично.
	\end{proof}
\end{statement}
\begin{statement}
	Пусть $A,B\in M_n$. Тогда $\det(AB)=\det A\cdot \det B$.
	\begin{proof}
		Применим цепочку элементарных преобразований строк, приводящую $A$
		к улучшенному ступенчатому виду $C$, то есть $C=U_kU_{k-1}\ldots
		U_1A$. При этом $\det C=\alpha\cdot\det A, \alpha\neq0$. Применим
		ту же цепочку преобразований к $AB$, получим $CB$, причём
		$\det(CB)=\alpha\cdot\det(AB)$. Пусть последняя строка $C_{(n)}$
		нулевая, тогда $(CB)_{(n)}$ также нулевая, то есть $\det
		C=0\implies \det (CB)=0=\det C\cdot \det B$. Пусть последняя строка
		ненулевая, тогда $C=E\implies CB=B\implies \det(CB)=\det B=\det
		C\cdot \det B$. В обоих случаях $\det(CB)=\det C\cdot \det
		B\implies \alpha\cdot \det(AB)=\alpha\cdot\det A\cdot \det
		B\implies \det(AB)=\det A\cdot \det B$.
	\end{proof}
\end{statement}
\begin{definition}
	Пусть $A\in M_n$. Дополнительный минор $\bar{M}_{ij}$ к
	элементу $a_{ij}$ -- это
	определитель матрицы размера $(n-1)\times (n-1)$, полученной из $A$
	вычёркиванием $i$-й строки и $j$-го столбца.
\end{definition}
\begin{definition}
	Пусть $A\in M_n$. Алгебраическое дополнение к элементу $a_{ij}$ --
	это величина:
	$$A_{ij}=(-1)^{i+j}\cdot \bar{M}_{ij}$$
\end{definition}
\begin{lemma}
	Пусть $A\in M_n$ и $\forall k\neq j\ a_{ik}=0$, тогда $\det
	A=a_{ij}\cdot A_{ij}$.
	\begin{proof}
		Переставляем соседние строки $i-1$ раз до тех пор, пока $i$-я
		строка не станет первой. После этого переставляем соседние столбцы
		$j-1$ раз до тех пор, пока $j$-й столбец не станет первым:
		$$A=
		\begin{pmatrix}
			P & U & Q\\
			0\ldots0 & a_{ij} & 0\ldots0\\
			R & V & S\\
		\end{pmatrix}\sim A'=
		\begin{pmatrix}
			0\ldots0 & a_{ij} & 0\ldots0\\
			P & U & Q\\
			R & V & S\\
		\end{pmatrix}\sim A''=
		\begin{pmatrix}
			a_{ij} & 0\ldots0 & 0\ldots0\\
			U & P & Q\\
			V & R & S\\
		\end{pmatrix}$$
		Тогда по утверждению об определителе с углом нулей:
		$$\det A''=a_{ij}\cdot\det
		\begin{pmatrix}
			P & Q\\
			R & S\\
		\end{pmatrix}=a_{ij}\cdot \bar{M}_{ij}$$
		Тогда, учитывая количество перестановок строк, получаем:
		$$\det A=(-1)^{i-1+j-1}\cdot\det A''=(-1)^{i+j}\cdot a_{ij}\cdot
		\bar{M}_{ij}=a_{ij}\cdot A_{ij}$$
	\end{proof}
\end{lemma}
\begin{theorem}[Разложение определителя по строке или столбцу]
	$$\det A=\sum_{j=1}^{n}a_{ij}\cdot A_{ij}=\sum_{i=1}^{n}a_{ij}\cdot A_{ij}$$
	\begin{proof}
		$$A_{(i)}=
		\begin{pmatrix}
			a_{i1} & a_{i2} & \ldots & a_{in}
		\end{pmatrix}=
		\begin{pmatrix}
			a_{i1} & 0 & \ldots & 0
		\end{pmatrix}+
		\begin{pmatrix}
			0 & a_{i2} & \ldots & 0
		\end{pmatrix}+\ldots+
		\begin{pmatrix}
			0 & 0 & \ldots & a_{in}
		\end{pmatrix}$$
		Тогда, разложив определитель по строкам, получаем:
		$$\det A=
		\begin{vmatrix}
			*\\
			a_{i1}\;0\;\ldots\; 0\\
			*\\
		\end{vmatrix}+
		\begin{vmatrix}
			*\\
			0\;a_{i2}\;\ldots\; 0\\
			*\\
		\end{vmatrix}+\ldots+
		\begin{vmatrix}
			*\\
			0\;0\;\ldots\; a_{in}\\
			*\\
		\end{vmatrix}=\sum_{j=1}^{n}a_{ij}\cdot A_{ij}$$
		Для столбцов также верно, так как $\det A=\det A^T$.
	\end{proof}
\end{theorem}
\begin{lemma}[О фальшивом разложении определителя]
	$$\forall i,k\in\{1,\ldots,n\},i\neq k\ \sum_{j=1}^{n}a_{ij}\cdot
	A_{kj}=0\quad \forall j,m\in\{1,\ldots,n\},j\neq
	m\ \sum_{i=1}^{n}a_{ij}\cdot A_{im}=0$$
	\begin{proof}
		Пусть $B$ -- матрица, полученная из $A$ заменой $k$-й строки на
		$i$-ю, тогда $\forall j\ b_{kj}=a_{ij}\implies
		B_{(i)}=B_{(k)}\implies \det B=0$. С другой стороны, разлагая по
		$k$-й строке, получаем:
		$$\det B=0=\sum_{j=1}^{n}b_{kj}\cdot B_{kj}=\sum_{j=1}^{n}a_{kj}\cdot A_{kj}$$
		Для столбцов также верно, так как $\det A=\det A^T$.
	\end{proof}
\end{lemma}
\begin{definition}
	Матрица $A$ называется вырожденной, если $\det A=0$.
\end{definition}
\begin{definition}
	Матрица $\widehat{A}=(A_{ij})^T$ называется присоединённой к $A$.
\end{definition}
\begin{theorem}[Формула Крамера]
	Пусть дана СЛУ $Ax=b, A\in M_n,\ x, b\in\R^n$. Тогда если $\det
	A\neq0$, то СЛУ имеет единственное решение вида $x_i=\frac{\det
	A_i}{\det A}$, где $A_i$ -- матрица, полученная из $A$ заменой
	$i$-го столбца на $b$.
	\begin{proof}
		Поскольку $\det A\neq0$, у неё есть обратная. Умножим СЛУ слева на
		$A^{-1}$, получая $x=A^{-1}b$, откуда видно, что решение
		единственно. При этом $A^{-1}=\frac{\widehat{A}}{\det A}$, тогда
		$x=\frac{\widehat{A}}{\det A}b$. Получаем, что:
		$$\det A\cdot x_i=(\widehat{A})_{(i)}b=\sum_{j=1}^{n}A_{ji}b_j=\det A_i$$
	\end{proof}
\end{theorem}

\subsection{Обратные матрицы}

\begin{definition}
	Пусть $A\in M_n$. Матрица $B=A^{-1}\in M_n$ называется обратной к
	$A$, если $AB=BA=E$.
\end{definition}
\begin{lemma}
	Если у матрицы существует обратная, то она единственная.
	\begin{proof}
		Пусть $B, B'$ -- обе обратные к $A$, тогда:
		$$B'=EB'=(BA)B=B(AB')=BE=B$$
	\end{proof}
\end{lemma}
\begin{lemma}\label{Если обратима, то невырождена}
	Если $A$ обратима, то $\det A\neq0$ и $\det A^{-1}=\frac{1}{\det A}$.
	\begin{proof}
		$$AA^{-1}=E\implies \det(AA^{-1})=\det E=1\implies \det A\cdot\det A^{-1}=1$$
	\end{proof}
\end{lemma}
\begin{theorem}
	Матрица $A$ обратима тогда и только тогда, когда $A$ невырождена,
	при этом:
	$$A^{-1}=\frac{1}{\det A}\cdot \widehat{A}$$
	\begin{proof}
		Следствие в одну сторону верно из леммы \ref{Если обратима, то
		невырождена}. Пусть $\det A\neq0$, покажем, что
		$A\cdot\frac{\widehat{A}}{\det A}=\frac{\widehat{A}}{\det A}\cdot
		A=E$. Для матрицы $X=A\cdot \widehat{A}$ имеем:
		$$x_{ij}=\sum_{k=1}^{n}a_{ik}\cdot(\widehat{A})_{kj}=\sum_{k=1}^{n}a_{ik}\cdot
		A_{jk}=
		\begin{+cases}
			\det A,\ i=j\\
			0,\ i\neq j
		\end{+cases}$$
		Аналогично для $Y=\widehat{A}\cdot A$ имеем:
		$$y_{ij}=\sum_{k=1}^{n}(\widehat{A})_{ik}\cdot
		a_{kj}=\sum_{k=1}^{n}a_{kj}\cdot A_{ki}=
		\begin{+cases}
			\det A,\ i=j\\
			0,\ i\neq j
		\end{+cases}$$
		То есть, $A\widehat{A}=\widehat{A}A=\det A\cdot E$.
	\end{proof}
\end{theorem}
\begin{consequence}
	Если $A, B\in M_n$ и $AB=E$, то $BA=E$.
	\begin{proof}
		$$AB=E\implies \det A\cdot \det B=1\implies \det A\neq0\implies
		\exists\,A^{-1}$$
		$$BA=EBA=(A^{-1}A)(BA)=A^{-1}(AB)A=A^{-1}A=E$$
	\end{proof}
\end{consequence}
\begin{consequence}
	Пусть $A, B\in M_n$, тогда $AB$ обратима тогда и только тогда, когда
	и $A$, и $B$ обратимы, при этом $(AB)^{-1}=B^{-1}A^{-1}$.
	\begin{proof}
		Первая часть следует напрямую из теоремы. Остаётся заметить, что
		$$(AB)(B^{-1}A^{-1})=(B^{-1}A^{-1})(AB)=E$$
	\end{proof}
\end{consequence}
