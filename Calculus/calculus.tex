\section{Аксиоматика действительных чисел}

\begin{axiom}[Аксиомы сложения]
	\hfill
	\begin{center}
		\begin{minipage}{0.4\textwidth}
			\begin{enumerate}[itemsep=0pt]
				\item $\forall a, b\in \R\ a+b=b+a$;
				\item $\forall a, b, c \in \R\ (a+b)+c=a+(b+c)$;
				\item $\exists\,0\in\R\colon \forall a\in \R\ a+0=a$;
				\item $\forall a\in \R\ \exists\,(-a)\in\R\colon a+(-a)=0$.
			\end{enumerate}
		\end{minipage}
	\end{center}
\end{axiom}
\begin{axiom}[Аксиомы умножения]
	\hfill
	\begin{center}
		\begin{minipage}{0.4\textwidth}
			\begin{enumerate}[itemsep=0pt, start=5]
				\item $\forall a, b\in \R\ a\cdot b=b\cdot a$;
				\item $\forall a, b, c \in \R\ (a\cdot b)\cdot c=a\cdot(b\cdot c)$;
				\item $\exists\,1\in\R\colon \forall a\in \R\ a\cdot 1=a$;
				\item $\forall a\in
					\R\setminus\{0\}\ \exists\,\frac{1}{a}\in\R\colon a\cdot\frac{1}{a}=1$.
			\end{enumerate}
		\end{minipage}
	\end{center}
\end{axiom}
\begin{axiom}[Аксиома связи сложения и умножения]
	\hfill
	\begin{center}
		\begin{minipage}{0.4\textwidth}
			\begin{enumerate}[itemsep=0pt, start=9]
				\item $\forall a, b, c \in \R\ a\cdot (b+c)=a\cdot b+a\cdot c$.
			\end{enumerate}
		\end{minipage}
	\end{center}
\end{axiom}
\begin{axiom}[Аксиомы отношения порядка]
	\hfill
	\begin{center}

				\item $\forall a, b\in \R\ (a\leqslant b \land b\leqslant a)\implies a=b$;

\begin{axiom}[Аксиома непрерывности]
	\hfill
	\begin{center}

				\item $A, B \subset \R, A, B \neq \varnothing, \forall

	\end{center}
\end{axiom}

\begin{definition}
	Вещественное число можно определить как бесконечную десятичную дробь:
	\[\pm a_0,a_1a_2\ldots a_n\ldots\]
	Где $a_0$ -- целое неотрицательное число, $a_1,a_2,\ldots,
	a_n,\ldots$ -- последовательность десятичных знаков. При этом
	$a_0,a_1a_2\ldots a_n(9)=a_0,a_1a_2\ldots a_n+10^{-n}$, поэтому
	запись с 9 в периоде всегда заменяется эквивалентной с нулём в
	периоде. Сравнение вещественных чисел в такой форме производится поразрядно.
\end{definition}
\begin{theorem}
	Для вещественных чисел в форме бесконечной десятичной дроби
	выполняется аксиома непрерывности.
	\begin{proof}
		Если $\forall a\in A, b\in B\ a\leqslant 0, b\geqslant 0\implies c=0$.\par

		Если $\exists\, a\in A\colon a>0$, то $\forall b\in B\ b>0$.
		Рассмотрим $c_0=\min\{b_0\mid \exists\, b\in B\colon
		b=b_0\ldots\}$, $c_1=\min\{b_1\mid \exists\, b\in B\colon b=c_0b_1\ldots\}$,
		$c_k=\min\{b_k\mid \exists\, b\in B\colon b=c_0c_1\ldots
		c_{k-1}b_k\ldots\}$. Пусть $c=c_0c_1\ldots c_k\ldots$. Тогда $\forall a\in
		A, b\in B\ a\leqslant c\leqslant b$ по построению.\par

		Если $\exists\, b\in B\colon b<0$, то рассмотрим $A'=-B, B'=-A$,
		далее аналогично предыдущему случаю.
	\end{proof}
\end{theorem}
\begin{statement}[Неравенство треугольника]
	\[\abs{a+b}\leqslant \abs{a}+\abs{b}\]
\end{statement}
\begin{consequence}
	\[\abs{a}-\abs{b}\leqslant \abs{a-b}\]
	\begin{proof}
		Пусть $a'=a-b, b'=b$, тогда:
		\[\abs{a'+b'}\leqslant \abs{a'}+\abs{b'} \implies
			\abs{(a-b)+b}\leqslant \abs{a-b}+\abs{b}\implies
		\abs{a}-\abs{b}\leqslant \abs{a-b}\]	
	\end{proof}
\end{consequence}

\section{Пределы}

\subsection{Верхняя и нижняя грани}

\begin{definition}
	Число $M\in \R$ называется верхней гранью $A\subset \R$, если
	$\forall a\in A\ a\leqslant M$.
\end{definition}
\begin{definition}
	Множество $A\subset \R$ называется ограниченным сверху, если:
	\[\exists\,M\in\R\colon \forall a\in A\ a\leqslant M\]
\end{definition}
\begin{definition}
	Число $M\in \R$ называется нижней гранью $A\subset \R$, если
	$\forall a\in A\ a\geqslant M$.
\end{definition}
\begin{definition}
	Множество $A\subset \R$ называется ограниченным снизу, если:
	\[\exists\,M\in\R\colon \forall a\in A\ a\geqslant M\]
\end{definition}
\begin{definition}
	Множество $A\subset \R$ называется ограниченным, если оно ограничено
	сверху и ограничено снизу.
\end{definition}
\begin{definition}
	Число $M$ называется максимальным элементом $A\subset \R$, если $M$
	является верхней гранью $A$ и $M\in A$.
\end{definition}
\begin{definition}
	Число $M$ называется минимальным элементом $A\subset \R$, если $M$
	является нижней гранью $A$ и $M\in A$.
\end{definition}
\begin{definition}
	Число $M\in \R$ называется точной верхней гранью (или супремумом)
	$A\subset \R$, если:
	\[M=\sup A\iff (\forall a\in A\ a\leqslant M) \land (\forall M'<M\
	\exists\,a\in A\colon a>M')\]
\end{definition}
\begin{definition}
	Число $M\in \R$ называется точной нижней гранью (или инфимумом)
	$A\subset \R$, если:
	\[M=\inf A\iff (\forall a\in A\ a\geqslant M) \land (\forall M'>M\
	\exists\,a\in A\colon a<M')\]
\end{definition}
\begin{theorem}
	Пусть множество $A\subset\R, A\neq\varnothing$ ограничено сверху,
	тогда $\exists!\,M\colon M=\sup A$.
	\begin{proof}
		Пусть $B$ -- множество всех верхних граней $A$. По аксиоме
		непрерывности $\exists\,c\in \R\colon \forall a\in A, b\in B\ a\leqslant
		c\leqslant b$. Пусть $\exists\,M'<c\colon \forall a\in A\ a\leqslant M'$.
		$M'$ -- верхняя грань, то есть $M'\in B\implies c\leqslant M'$, но
		$M'<c$ по предположению, противоречие.
	\end{proof}
\end{theorem}
\begin{theorem}[Принцип Архимеда]
	\[\forall x\in \R\ \exists\,n\in\N\colon n>x\]
	\begin{proof}
		Пусть $\exists\,x\in\R\colon \forall n\in \N\ n\leqslant x$. Тогда
		$\exists\,c=\sup\N$, то есть $\forall n\ n\leqslant c,
		\forall\varepsilon>0\ \exists\,n\colon n>c-\varepsilon$. Рассмотрим
		$\varepsilon=1$, тогда $n>c-1$, значит $n+1>c$, но $(n+1)\in\N$, то
		есть $c$ -- не точная верхняя грань.
	\end{proof}
\end{theorem}
\begin{consequence}[Целая часть числа]
	\[\forall x\in\R\ \exists!\,m\in\Z\colon m\leqslant x < m+1\]
	\begin{proof}
		Рассмотрим $x\geqslant0$. Пусть $c=\inf\{n\in\N\colon n>x\}$, тогда
		$c\in\N$, и $m=c-1$. При $x<0$ аналогично.
	\end{proof}
\end{consequence}
\begin{consequence}[Плотность $\Q$ в $\R$]
	\[\forall a,b\in\R\colon a<b\ \exists\, q\in\Q\colon a<q<b\]
	\begin{proof}
		По принципу Архимеда:
		\[\exists\,n\in\N\colon n>\frac{1}{b-a}\implies \frac{1}{n}<b-a\]
		В силу существования целой части числа:
		\[\exists\,k\in\Z\colon  k-1\leqslant an < k\implies
		\frac{k-1}{n}\leqslant a<\frac{k}{n}\]		Пусть $q=\frac{k}{n}$,
		тогда $q\in\Q$ по определению, по построению получаем:
		\[a\geqslant \frac{k-1}{n}=q-\frac{1}{n}\]
		Предположим $q\geqslant b$, тогда:
		\[b-a\geqslant q-\left(q-\frac{1}{n}\right)=\frac{1}{n}\]
		Противоречие, то есть $a<q<b$.
	\end{proof}
\end{consequence}

\subsection{Пределы числовых последовательностей}

\begin{definition}
	Числовой последовательностью $\{a_n\}$ называется функция $a\colon
	\N\to\R$, где $\forall n\in \N\ a(n)=a_n$. Элемент
	последовательности -- это пара $(n, a_n)$.
\end{definition}
\begin{definition}
	Пусть $a,\varepsilon \in \R, \varepsilon>0$. Интервал
	$B_\varepsilon(a)=(a-\varepsilon, a+\varepsilon)$ называется
	$\varepsilon$-окрестностью числа $a$.
\end{definition}
\begin{definition}
	Число $a\in \R$ называется пределом последовательности $\{a_n\}$, если:
	\[a=\lim_{n\to\infty}a_n\iff
		\forall\varepsilon>0\ \exists\,N(\varepsilon)\colon \forall n\geqslant
	N(\varepsilon)\ a_n\in B_\varepsilon(a)\]
\end{definition}
\begin{statement}
	Пусть $\alpha>0$ -- фиксированное число, тогда:
	\[a=\lim_{n\to\infty}a_n\iff
		\forall\varepsilon>0\ \exists\,N(\varepsilon)\colon \forall n>
	N(\varepsilon)\ \abs{a_n-a}\leqslant \alpha\varepsilon\]
\end{statement}
\begin{statement}[Единственность предела]
	\[\lim_{n\to\infty}a_n=a, \lim_{n\to\infty}a_n=b \implies a=b\]
	\begin{proof}
		Пусть $a>b$, тогда $a-b=\varepsilon_0>0$. Рассмотрим
		$\varepsilon=\frac{\varepsilon_0}{2}$, тогда:
		\[\exists\,N_1\colon \forall n\geqslant N_1\ \abs{a_n-a}<\varepsilon, \qquad
			\exists\,N_2\colon \forall n\geqslant
		N_2\ \abs{a_n-b}<\varepsilon\]		\[\forall n\geqslant \max(N_1,
		N_2)\ \varepsilon_0=\abs{a-b}\leqslant\abs{a_n-a}+\abs{a_n-b}<\frac{\varepsilon_0}{2}+\frac{\varepsilon_0}{2}=\varepsilon_0\]	
	\end{proof}
\end{statement}
\begin{definition}
	Последовательность $\{a_n\}_{n=1}^\infty$ называется ограниченной, если:
	\[\exists\, c, C\in \R\colon\forall n\in \N\  c\leqslant a_n\leqslant C\]
\end{definition}
\begin{definition}
	Последовательность называется сходящейся, если у неё есть предел.
\end{definition}
\begin{statement}
	Сходящаяся последовательность ограничена.
	\begin{proof}
		\[a=\lim_{n\to\infty}a_n\iff \forall \varepsilon >0\ \exists\,
		N\colon \forall n\geqslant N\ \abs{a_n-a}<\varepsilon\]		Пусть
		$\varepsilon=1$, тогда $\abs{a_n-a}<1$ и
		$\abs{a_n-a}\geqslant\abs{a_n}-\abs{a}$,
		то есть $\abs{a_n}<1+\abs{a}$. Тогда возьмем:
		\[C=\max\{a_1,a_2,\ldots,a_{N-1}, 1+\abs{a}\}, \qquad
		c=\min\{a_1,a_2,\ldots,a_{N-1}, -(1+\abs{a})\}\]		Получаем, что
		$\forall n\in \N\ c\leqslant a_n \leqslant C$, то есть
		последовательность ограничена.
	\end{proof}
\end{statement}
\begin{lemma}[Об отделимости]
	\[a_n\to a, a\neq0 \implies \exists\, N\in \N\colon \forall
	n>N\ \abs{a_n}>\frac{\abs{a}}{2}>0\]	
	\begin{proof}
		Пусть $\varepsilon=\frac{\abs{a}}{2}>0$. Тогда:
		$\exists\, N\in \N\colon
		\forall n>N\ \abs{a_n-a}<\frac{\abs{a}}{2}$, но
		$\abs{a}-\abs{a_n}\leqslant\abs{a_n-a}<\frac{\abs{a}}{2}$, то есть
		$\abs{a_n}>\frac{\abs{a}}{2}$.
	\end{proof}
\end{lemma}
\begin{theorem}[Арифметические свойства предела последовательности]
	Пусть $\lim\limits_{n\to\infty}a_n=a, \lim\limits_{n\to\infty}b_n=b$, тогда:
	\begin{center}
		\begin{minipage}{0.35\textwidth}
			\begin{enumerate}[itemsep=0pt]
				\item $\lim\limits_{n\to\infty}(\alpha a_n+\beta b_n)=\alpha a+\beta b$;
				\item $\lim\limits_{n\to\infty}a_nb_n=ab$;
				\item $\lim\limits_{n\to\infty}\dfrac{a_n}{b_n}=\dfrac{a}{b}, b\neq
					0, b_n\neq 0$.
			\end{enumerate}
		\end{minipage}
	\end{center}
	\begin{proof}
		\hfill
		\begin{enumerate}
			\item Докажем, что предел линейной комбинации равен линейной
				комбинации пределов.
				\[\abs{\alpha a_n+\beta b_n-(\alpha a+\beta
					b)}=\abs{\alpha(a_n-a)+\beta(b_n-b)}\leqslant
				\abs{\alpha}\cdot\abs{a_n-a}+\abs{\beta}\cdot\abs{b_n-b}\]				Тогда
				воспользуемся
				определением предела при $\gamma=\abs{\alpha}+\abs{\beta}$:
				\[\forall \varepsilon>0\ \exists\, N\colon \forall
					n>N\ \abs{\alpha a_n+\beta b_n-(\alpha a+\beta b)}\leqslant
				\gamma\varepsilon\]			
			\item Докажем, что предел произведения равен произведению пределов.
				\[\abs{a_nb_n}\leqslant \abs{a_nb_n-a_nb} +
				\abs{a_nb-ab}=\abs{a_n}\cdot\abs{b_n-b}+\abs{b}\cdot\abs{a_n-a}\]				$a_n$
				и $b_n$
				сходятся, а значит ограничены, то есть для
				фиксированного $\varepsilon$ верно:
				\[\abs{a_n}\leqslant M, \abs{b_n-b}<\varepsilon,
					\abs{a_n-a}<\varepsilon\implies
					\abs{a_n}\cdot\abs{b_n-b}+\abs{b}\cdot\abs{a_n-a}\leqslant
				M\varepsilon+\abs{b}\varepsilon=(M+\abs{b})\varepsilon\]				Тогда
				воспользуемся определением предела при $\alpha=M+\abs{b}$:
				\[\forall \varepsilon>0\ \exists\, N\colon \forall
				n>N\ \abs{a_nb_n-ab}\leqslant \alpha\varepsilon\]			
			\item Докажем, что предел отношения равен отношению пределов.
				\[\abs{\frac{1}{b_n}-\frac{1}{b}}=\abs{\frac{b-b_n}{b\cdot
				b_n}}=\frac{\abs{b-b_n}}{\abs{b}\cdot\abs{b_n}}\]				По
				лемме об отделимости:
				\[\exists\, N\in \N\colon \forall
				n>N\ \abs{b_n}>\frac{\abs{b}}{2}\]				Отсюда получаем предел $b_n^{-1}$
				при $\varepsilon=\frac{b^2}{2}$:
				\[\frac{\abs{b-b_n}}{\abs{b}\cdot\abs{b_n}}<\frac{2\abs{b-b_n}}{b^2}<\frac{2\varepsilon}{b^2}\]
				Тогда предел отношения $a_n$ к $b_n$ получается умножением
				пределов $a_n$ и $b_n^{-1}$, что уже было
				доказано в предыдущем пункте.
		\end{enumerate}
	\end{proof}
\end{theorem}
\begin{lemma}[О переходе к пределу в неравенстве]
	Пусть $\lim\limits_{n\to\infty}a_n=a,
	\lim\limits_{n\to\infty}b_n=b$, тогда:
	\[\exists\,N\colon \forall
	n>N\ a_n\leqslant b_n\implies a\leqslant b\]	
	\begin{proof}
		Пусть $a>b$, $a-b=\varepsilon_0>0$,  $N$ -- номер, для которого:
		\[\forall n>N\ \abs{a-a_n}<\frac{\varepsilon_0}{2},
		\abs{b-b_n}<\frac{\varepsilon_0}{2}\]		\[\varepsilon_0=a-b=a-a_n+a_n-b_n+b_n-b\leqslant
		a-a_n+b_n-b\leqslant \abs{a-a_n}+\abs{b_n-b}<\varepsilon_0\]	
	\end{proof}
\end{lemma}
\begin{lemma}[О зажатой последовательности]
	Пусть $\lim\limits_{n\to\infty}a_n=\lim\limits_{n\to\infty}b_n=a$, тогда
	\[\exists\,N\colon \forall n>N\ a_n\leqslant c_n\leqslant
	b_n\implies \lim\limits_{n\to\infty}c_n=a\]	
	\begin{proof}
		Для фиксированного $\varepsilon$ справедливо:
		\[\abs{a_n-a}<\varepsilon\iff
			\begin{+cases}
				a_n-a<\varepsilon\\
				-a_n+a<\varepsilon
			\end{+cases}\qquad
		b_n<a+\varepsilon\]		Тогда есть $N$, для которого:
		\[a_n-\varepsilon<a_n\leqslant c_n \leqslant b_n < a+\varepsilon\]
	\end{proof}
\end{lemma}
\begin{theorem}[Вейерштрасс]
	Пусть последовательность $\{a_n\}_{n=1}^\infty$ не убывает и
	ограничена сверху, тогда:
	\[\lim_{n\to\infty}a_n=\sup a_n\]
	Аналогичное утверждение верно для инфимума.
	\begin{proof}
		Пусть $M=\sup a_n$, он существует, так как $a_n$
		ограничена, тогда:
		\[(\forall n\ a_n\leqslant M)\land (\forall M'<M\ \exists\, n\colon a_n>M')\]
		Возьмём $M'=M-\varepsilon<M$, тогда $\exists\, N\colon
		a_N>M-\varepsilon\implies \forall n\geqslant N\  \abs{a_n-M}<\varepsilon$.
	\end{proof}
\end{theorem}
\begin{definition}
	Последовательность $\{a_n\}$ называется бесконечно малой, если
	$\lim\limits_{n\to\infty}=0$.
\end{definition}
\begin{definition}
	Последовательность $\{a_n\}$ называется бесконечно большой, если:
	\[\forall c>0\ \exists\, N\colon \forall n>N\ \abs{a_n}>c\]
\end{definition}

\begin{statement}
	\[\forall k \in \N\ k!\geqslant2^{k-1}\]
	\begin{proof}
		Заметим, что для $k=1$ верно: $1!=1\geqslant 2^0$. Пусть для $n$
		верно, рассмотрим для $n+1$:
		\[(k+1)!=k!\cdot(k+1)\geqslant 2^{k-1}\cdot(k+1)\geqslant 2^{k-1}\cdot2=2^k\]
	\end{proof}
\end{statement}

\begin{theorem}
	\[\exists\,a\colon a=\lim_{n\to\infty}\left(1+\frac{1}{n}\right)^n\]
	\begin{proof}
		Покажем, что последовательность ограничена сверху:
		\begin{multline*}
			a_n=\left(1+\frac{1}{n}\right)^n=\sum_{k=0}^{n}C_n^k\frac{1}{n^k}=2+\sum_{k=2}^{n}\frac{n!}{k!\cdot(n-k)!}\cdot\frac{1}{n^k}=\\
			=2+\sum_{k=2}^{n}\frac{1}{k!}\left(\frac{n}{n}\cdot\frac{n-1}{n}\cdot\ldots\cdot\frac{n-(k-1)}{n}\right)=2+\sum_{k=2}^{n}\frac{1}{k!}\left(1-\frac{1}{n}\right)\cdot\left(1-\frac{2}{n}\right)\cdot\ldots\cdot\left(1-\frac{k-1}{n}\right)\leqslant\\
			\leqslant2+\sum_{k=2}^{n}\frac{1}{k!}\leqslant2+\sum_{k=2}^{n}\frac{1}{2^{k-1}}\leqslant2+1=3
		\end{multline*}
		Покажем, что последовательность не убывает:
		\begin{multline*}
			a_n=2+\sum_{k=2}^{n}\frac{1}{k!}\left(1-\frac{1}{n}\right)\cdot\left(1-\frac{2}{n}\right)\cdot\ldots\cdot\left(1-\frac{k-1}{n}\right)\leqslant\\
			\leqslant2+\sum_{k=2}^{n}\frac{1}{k!}\left(1-\frac{1}{n+1}\right)\cdot\left(1-\frac{2}{n+1}\right)\cdot\ldots\cdot\left(1-\frac{k-1}{n+1}\right)\leqslant\\
			\leqslant2+\sum_{k=2}^{n+1}\frac{1}{k!}\left(1-\frac{1}{n+1}\right)\cdot\left(1-\frac{2}{n+1}\right)\cdot\ldots\cdot\left(1-\frac{k}{n+1}\right)=a_{n+1}
		\end{multline*}
		Тогда по теореме Вейерштрасса данная последовательность сходится.
	\end{proof}
\end{theorem}
\begin{theorem}[Принцип вложенных отрезков]
	Всякая последовательность $\{[a_n,b_n]\}_{n=1}^\infty$ вложенных
	отрезков имеет общую точку. Кроме того, если $b_n-a_n\to0$, то такая
	точка только одна.
	\begin{proof}
		Рассмотрим $A=\{a_n\}_{n=1}^\infty,B=\{b_n\}_{n=1}^\infty$. По условию:
		\[\forall n\ (a_n\leqslant b_n) \land (a_{n+1}\geqslant a_n) \land
		(b_{n+1}\leqslant b_n)\]		Заметим, $\forall a\in A, b\in
		B\ a\leqslant b$, тогда по аксиоме
		непрерывности:
		\[\exists\,c\colon \forall a\in A, b\in B\ a\leqslant c\leqslant
		b\implies \forall n\ c\in[a_n, b_n]\]		Теперь пусть $b_n-a_n\to0$,
		$\exists\,c'\neq c\colon \forall
		n\ c'\in [a_n,b_n]$. Пусть $c'>c$, тогда $c'-c\leqslant
		b_n-a_n\to0$, значит $c=c'$, для $c'<c$ аналогично.
	\end{proof}
\end{theorem}
\begin{definition}
	Пусть задана последовательность $\{a_n\}_{n=1}^\infty$ и пусть
	задана возрастающая последовательность натуральных чисел
	$n_1<n_2<n_3<\ldots$ Последовательность $b_k=a_{n_k}$ называется
	подпоследовательностью последовательности $\{a_n\}_{n=1}^\infty$.
\end{definition}
\begin{definition}
	Число $a\in\R$ называется частичным пределом последовательности
	$\{a_n\}_{n=1}^\infty$, если для некоторой подпоследовательности
	$\{a_{n_k}\}_{k=1}^\infty$ оно является пределом.
\end{definition}
\begin{statement}
	Любая подпоследовательность сходящейся последовательности сходится к
	пределу этой последовательности.
	\begin{proof}
		По определению предела:
		\[a=\lim_{n\to\infty}a_n\iff\forall\varepsilon>0\ \exists\,N_1(\varepsilon)\colon
			\forall n\geqslant
		N_1(\varepsilon)\ \abs{a_n-a}<\varepsilon\]		Заметим, что $n_k\geqslant
		k$. Для $k=1$ верно: $n_1\geqslant 1$,
		пусть для $k$ верно, рассмотрим $k+1$: $n_{k+1}\geqslant n_k +
		1\geqslant k+1$. Тогда получаем, что:
		\[\forall\varepsilon>0\ \exists\,N_1(\varepsilon)\colon
		\forall k\geqslant N_1(\varepsilon)\ \abs{a_{n_k}-a}<\varepsilon\]	
	\end{proof}
\end{statement}
\begin{definition}
	Пусть $\{a_n\}_{n=1}^\infty$ -- ограниченная последовательность.
	Рассмотрим последовательности $M_n\coloneq\sup\limits_{k>n} a_k$ и
	$m_n\coloneq\inf\limits_{k>n} a_k$. Последовательность $M_n$ не возрастает и
	ограничена снизу, а последовательность $m_n$ не убывает и ограничена
	сверху, тогда по теореме Вейерштрасса существуют верхний и нижний
	частичные пределы последовательности $\{a_n\}_{n=1}^\infty$:
	\[\limsup_{n\to\infty}a_n\coloneq\lim_{n\to\infty} M_n, \qquad
	\liminf_{n\to\infty}a_n\coloneq\lim_{n\to\infty} m_n\]
\end{definition}
\begin{theorem}
	Пусть $\{a_n\}_{n=1}^\infty$ -- ограниченная последовательность.
	Тогда $\limsup\limits_{n\to\infty}a_n$ и
	$\liminf\limits_{n\to\infty}a_n$ -- частичные пределы
	последовательности $\{a_n\}_{n=1}^\infty$, и любой другой частичный
	предел принадлежит отрезку:
	\[\left[\liminf_{n\to\infty}a_n, \limsup_{n\to\infty}a_n\right]\]
	\begin{proof}
		Пусть $M_n=\sup\limits_{k>n} a_k,\ M=\lim\limits_{n\to\infty}M_n$.
		Положим $n_1=1$ и $n_1<n_2<\ldots<n_k$, возьмём такой $n_{k+1}>n_k$, что:
		\[M_{n_k}-\frac{1}{k}<a_{n_{k+1}}\leqslant M_{n_k}\]
		Тогда $M_{n_k}\to M$ как подпоследовательность сходящейся
		последовательности и $M_{n_k}-\frac{1}{k}\to M$. Значит по лемме о
		зажатой последовательности $a_{n_k}\to M$, то есть $M$ -- частичный
		предел $a_n$. Аналогично доказывается для нижнего предела.\par

		Пусть $a$ -- частичный предел, то есть
		$a=\lim\limits_{k\to\infty}a_{n_k}$. Заметим, что:
		\[m_{n_{k-1}}\leqslant a_{n_k}\leqslant M_{n_{k-1}}\]
		Тогда по лемме о переходе к пределу в неравенстве $a$ лежит в
		отрезке из условия.
	\end{proof}
\end{theorem}
\begin{consequence}[Теорема Больцано-Вейерштрасса]
	Во всякой ограниченной последовательности можно найти сходящуюся
	подпоследовательность.
\end{consequence}
\begin{theorem}
	Ограниченная последовательность сходится тогда и только тогда, когда
	множество её частичных пределов состоит из одного элемента.
	\begin{proof}
		Если последовательность сходится, то все её подпоследовательности
		сходятся к тому же пределу, то есть множество её частичных пределов
		состоит из одного элемента.\par

		Пусть $a_n$ ограничена и $M_n\to a, m_n\to a$. Поскольку
		$m_{n-1}\leqslant a_n\leqslant M_{n-1}$, по лемме о зажатой
		последовательности $a_n\to a$, то есть сходится.
	\end{proof}
\end{theorem}
\begin{definition}
	Говорят, что последовательность $\{a_n\}_{n=1}^\infty$
	фундаментальна (или является последовательностью Коши), если:
	\[\forall \varepsilon>0\ \exists\,N(\varepsilon)\in\N\colon\forall
	n, m>N(\varepsilon)\  \abs{a_n-a_m}<\varepsilon\]
\end{definition}
\begin{theorem}[Критерий Коши]
	Числовая последовательность сходится тогда и только тогда, когда она
	фундаментальна.
	\begin{proof}
		Пусть $a_n\to a$, возьмём произвольный $\varepsilon>0$, тогда:
		\[\exists\,N\colon\forall n\geqslant N\ \abs{a_n-a}<\frac{\varepsilon}{2}\]
		Тогда получаем, что последовательность фундаментальна по определению:
		\[\forall n, m>N\ \abs{a_n-a_m}\leqslant
		\abs{a_n-a}+\abs{a_m-a}<\frac{\varepsilon}{2}+\frac{\varepsilon}{2}=\varepsilon\]		Пусть
		последовательность $a_n$ фундаментальна, возьмём $\varepsilon=1$, тогда:
		\[\forall n, m>N\ \abs{a_n-a_m}<1\]
		Рассмотрим $m=N+1$, тогда
		$1>\abs{a_n-a_{N+1}}\geqslant\abs{a_n}-\abs{a_{N+1}}\implies
		\abs{a_n}<\abs{a_{N+1}}+1$, то есть последовательность ограничена:
		\[\forall n\ \abs{a_n}\leqslant \max\{\abs{a_{N+1}}+1, \abs{a_N},
		\abs{a_{N-1}},\ldots,\abs{a_1}\}\]		Значит, по теореме
		Больцано-Вейерштрасса для фиксированного $\varepsilon>0$:
		\[\exists\,K\colon\forall k\geqslant
		K\ \abs{a_{n_k}-a}<\frac{\varepsilon}{2}\]
		Рассмотрим $N$ из условия фундаментальности $a_n$, тогда она сходится:
		\[\forall n\geqslant N\ \abs{a_n-a}\leqslant
		\abs{a_n-a_{n_k}}+\abs{a_{n_k}-a}<\varepsilon\]	
	\end{proof}
\end{theorem}

\subsection{Числовые ряды}

\begin{definition}
	Пусть $\{a_n\}_{n=1}^\infty$ -- числовая последовательность.
	Числовым рядом с членами $a_n$ называется выражение:
	\[\sum_{k=1}^{\infty}a_k\]
\end{definition}
\begin{definition}
	Конечные суммы $S_n\coloneq\sum\limits_{k=1}^{n}a_k$ называют
	частичными суммами ряда $\sum\limits_{k=1}^{\infty}a_k$.
\end{definition}
\begin{definition}
	Говорят, что ряд $\sum\limits_{k=1}^{\infty}a_k$ сходится, если у
	последовательности $\{S_n\}_{n=1}^\infty$ существует предел, который
	называют суммой ряда.
\end{definition}
\begin{theorem}[Критерий Коши]
	\[\text{Ряд }\sum\limits_{k=1}^{\infty}a_k\text{ сходится}\iff
		\forall\varepsilon>0\ \exists\, N\colon\forall
	n>m>N\ \abs{\sum_{k=m+1}^{n}a_k}=\abs{S_n-S_m}<\varepsilon\]	
	\begin{proof}
		\[\abs{S_n-S_m}=\abs{\sum_{k=1}^{n}a_k-\sum_{k=1}^{m}a_k}=\abs{\sum_{k=m+1}^{n}a_k}\]
	\end{proof}
\end{theorem}
\begin{consequence}
	Если ряд $\sum\limits_{k=1}^{\infty}a_k$ сходится, то
	$\lim\limits_{k\to\infty}a_k=0$.
	\begin{proof}
		По критерию Коши получаем:
		\[\forall\varepsilon>0\ \exists\, N\colon\forall
		n>m>N\ \abs{\sum_{k=m+1}^{n}a_k}=\abs{S_n-S_m}<\varepsilon\]		
		Пусть $m=n+1$, тогда
		$\abs{S_m-S_n}=\abs{S_{n+1}-S_n}=\abs{a_{n+1}}<\varepsilon$, то
		есть $a_k\to0$.
	\end{proof}
\end{consequence}
\begin{consequence}
	Из сходимости ряда $\sum\limits_{k=1}^{\infty}\abs{a_k}$ следует
	сходимость ряда $\sum\limits_{k=1}^{\infty}a_k$.
	\begin{proof}
		Из неравенства треугольника $\abs{a+b}\leqslant \abs{a}+\abs{b}$:
		\[\forall\varepsilon>0\ \exists\, N\colon\forall
		n>m>N\ \abs{\sum_{k=m+1}^{n}a_k}\leqslant\sum_{k=m+1}^{n}\abs{a_k}<\varepsilon\]	
	\end{proof}
\end{consequence}
\begin{definition}
	Говорят, что ряд $\sum\limits_{k=1}^{\infty}a_k$ сходится абсолютно,
	если сходится ряд $\sum\limits_{k=1}^{\infty}\abs{a_k}$.
\end{definition}
\begin{definition}
	Говорят, что ряд $\sum\limits_{k=1}^{\infty}a_k$ сходится условно,
	если он сходится, но ряд $\sum\limits_{k=1}^{\infty}\abs{a_k}$ расходится.
\end{definition}
\begin{statement}
	Пусть $a_k\geqslant0$, тогда ряд $\sum\limits_{k=1}^{n}a_k$ сходится тогда
	и только тогда, когда последовательность его частичных сумм ограничена.
	\begin{proof}
		$S_n$ не убывает, тогда по теореме Вейерштрасса она сходится тогда
		и только тогда, когда она ограничена.
	\end{proof}
\end{statement}
\begin{statement}[Признак сравнения]
	Пусть $0\leqslant a_n\leqslant b_n$. Если ряд
	$\sum\limits_{k=1}^{\infty}b_k$ сходится, то ряд
	$\sum\limits_{k=1}^{\infty}a_k$ сходится. Если ряд
	$\sum\limits_{k=1}^{\infty}a_k$ расходится, то ряд
	$\sum\limits_{k=1}^{\infty}b_k$ расходится.
	\begin{proof}
		Пусть $A_n=\sum\limits_{k=1}^{n}a_k$,
		$B_n=\sum\limits_{k=1}^{n}b_k$. Тогда $A_n\leqslant B_n$, при
		этом $A_n$ и $B_n$ не убывают. Если $B_n$ ограничена, то есть
		сходится, то $A_n$ тоже сходится. Если $A_n$ неограниченна, то есть
		расходится, тогда $B_n$ тоже расходится.
	\end{proof}
\end{statement}
\begin{theorem}[Признак Лобачевского-Коши]
	Пусть $\{a_n\}_{n=1}^\infty$ -- невозрастающая последовательность,
	$a_n\geqslant0$. Ряд $\sum\limits_{k=1}^{\infty}a_k$ сходится тогда и
	только тогда, когда сходится ряд $\sum\limits_{k=1}^{\infty}2^ka_{2^k}$.
	\begin{proof}
		\[\sum_{k=1}^{n}2^{k-1}a_{2^k}\leqslant\sum_{k=2}^{2^n}a_k\leqslant\sum_{k=1}^{n}2^ka_{2^k}\]
		Тогда аналогично признаку сравнения выполняется утверждение теоремы.
	\end{proof}
\end{theorem}
\begin{definition}
	Будем говорить, что ряд $\sum\limits_{j=1}^{\infty}\tilde{a}_j$
	получен перестановкой членов ряда $\sum\limits_{k=1}^{\infty}a_k$,
	если $\exists\,\{k_k\}_{j=1}^\infty$, задающая взаимно-однозначное
	преобразование множества $\N$ на себя, и такая, что $\forall
	j\in\N\ \tilde{a}_j=a_{k_j}$.
\end{definition}
\begin{theorem}
	Если ряд $\sum\limits_{j=1}^{\infty}\tilde{a}_j$ получен
	перестановкой членов абсолютно сходящегося ряда
	$\sum\limits_{k=1}^{\infty}a_k$, то ряд
	$\sum\limits_{j=1}^{\infty}\tilde{a}_j$ абсолютно сходится, и его
	сумма равна сумме ряда $\sum\limits_{k=1}^{\infty}a_k$.
	\begin{proof}
		Пусть $\forall j\ a_j\geqslant0$, $M_n=\max\{k_j\colon j\leqslant
		n\}$, тогда:
		\[\sum_{j=1}^{n}\tilde{a}_j=\sum_{j=1}^{n}a_{k_j}\leqslant
		\sum_{k=1}^{M_n}a_k\leqslant \sum_{k=1}^{\infty}a_k\]
		Тогда при $n\to\infty$ получаем, что
		$\sum\limits_{j=1}^{\infty}\tilde{a}_j\leqslant\sum\limits_{k=1}^{\infty}a_k$,
		но поскольку можем провести обратную перестановку слагаемых, также
		верно
		$\sum\limits_{k=1}^{\infty}a_k\leqslant\sum\limits_{j=1}^{\infty}\tilde{a}_j$,
		то есть при перестановке слагаемых с неотрицательными челнами сумма
		не меняется.\par

		Пусть $a_j$
		могут быть отрицательны, тогда рассмотрим $a_j^+=\max\{0, a_j\}$ и
		$a_j^-=\max\{0 -a_j\}$, тогда $a_j=a_j^+-a_j^-$, аналогично для
		$\tilde{a}_j$. Заметим. что ряды
		$\sum a_j^+$ и $\sum a_j^-$ сходятся, так как по условию сходится
		$\sum\abs{a_j}$, тогда ряды $\sum \tilde{a}_j^+$ и $\sum \tilde{a}_j^-$
		тоже сходятся, так как сходится $\sum\abs{a_{k_j}}$. Получаем, что ряд
		$\sum\tilde{a}_j$ тоже сходится, то есть:
		\[\sum_{j=1}^{\infty}\tilde{a}_j=\sum_{j=1}^{\infty}(\tilde{a}_j^+-\tilde{a}_j^-)=\sum_{j=1}^{\infty}\tilde{a}_j^+-\sum_{j=1}^{\infty}\tilde{a}_j^-=\sum_{j=1}^{\infty}a_j^+-\sum_{j=1}^{\infty}a_j^-=\sum_{j=1}^{\infty}(a_j^+-a_j^-)=\sum_{j=1}^{\infty}a_j\]
	\end{proof}
\end{theorem}
\begin{theorem}[Римана]
	Если ряд $\sum\limits_{k=1}^{\infty}a_k$ сходится условно, то
	$\forall x\in \R\cup\{+\infty, -\infty\}$ можно так переставить
	члены этого ряда, что частичные суммы полученного ряда
	$\sum\limits_{j=1}^{\infty}\tilde{a}_j$ будут стремиться к $x$ (то
	есть при $x\in \R$ ряд будет сходиться к $x$).
\end{theorem}

\subsection{Пределы функций}

\begin{definition}
	Окрестностью радиуса $\varepsilon>0$ точки $a\in\R$ называется множество:
	\[B_\varepsilon(a)\coloneq\{x\in\R\colon
	\abs{x-a}<\varepsilon\}=(a-\varepsilon,a+\varepsilon)\]
\end{definition}
\begin{definition}
	Проколотой окрестностью радиуса $\varepsilon>0$ точки $a\in\R$
	называется множество:
	\[B'_\varepsilon(a)\coloneq
	B_\varepsilon(a)\setminus\{a\}=(a-\varepsilon,a)\cup(a, a+\varepsilon)\]
\end{definition}
\begin{definition}
	Точка $a\in\R$ называется внутренней точкой множества $M$, если:
	\[\exists\,\varepsilon>0\colon B_\varepsilon(a)\subset M\]
\end{definition}
\begin{definition}
	Точка $a\in\R$ называется предельной точкой множества $M$, если:
	\[\forall \varepsilon>0\ B'_\varepsilon(a)\cap M\neq\varnothing\]
\end{definition}
\begin{definition}
	Точка $a\in\R$ называется граничной точкой множества $M$, если:
	\[\forall \varepsilon>0\ (B'_\varepsilon(a)\cap
	M\neq\varnothing)\land(B_\varepsilon(a)\cap(\R\setminus M)\neq\varnothing)\]
\end{definition}
\begin{statement}
	Точка $a$ является предельной для $M$ тогда и только
	тогда, когда есть сходящаяся последовательность $a_n\in M\setminus\{a\}$.
	\begin{proof}
		Пусть $a$ -- предельная точка $M$, тогда $\forall n\exists a_n\in
		B'_{1/n}(a)\cap M$. Тогда $a_n\in M\setminus\{a\}$ и $a_n\to a$.
		Пусть $a_n\in M\setminus\{a\}$ и $a_n\to a$, тогда:
		\[\forall\varepsilon>0\ \exists\,N\colon \forall
		n>N\ \abs{a_n-a}<\varepsilon\]
		То есть $a_{N+1}\in B'_\varepsilon(a)\cap M$, а значит $a$ --
		предельная точка.
	\end{proof}
\end{statement}
\begin{definition}[Предел функции по Гейне]
	Пусть функция $f$ определена в некоторой проколотой окрестности
	$B'(a)$ точки $a$, тогда:
	\[A=\lim_{x\to a}f(x)\iff \forall x_n\in B'(a)\colon
	\lim_{n\to\infty}x_n=a\ \lim_{n\to\infty}f(x_n)=A\]
\end{definition}
\begin{definition}[Предел функции по Коши]
	Пусть функция $f$ определена в некоторой проколотой окрестности
	$B'(a)$ точки $a$, тогда:
	\[A=\lim_{x\to a}f(x)\iff
		\forall\varepsilon>0\ \exists\,\delta>0\colon\forall x\in
	B'_\delta(a)\ \abs{f(x)-A}<\varepsilon\]
\end{definition}
\begin{definition}[Предел функции по множеству по Коши]
	Пусть функция $f$ определена на некотором множестве $D\subset\R$ и
	$a$ -- предельная точка $D$, тогда:
	\[A=\lim_{x\to a}f(x)\iff
		\forall\varepsilon>0\ \exists\,\delta>0\colon\forall x\in
	D\cap B'_\delta(a)\ \abs{f(x)-A}<\varepsilon\]
\end{definition}
\begin{definition}[Предел функции по множеству по Гейне]
	Пусть функция $f$ определена на некотором множестве $D\subset\R$ и
	$a$ -- предельная точка $D$, тогда:
	\[A=\lim_{x\to a}f(x)\iff \forall x_n\in
	D\setminus\{a\}\colon \lim_{n\to\infty}x_n=a\ \lim_{n\to\infty}f(x_n)=A\]
\end{definition}
\begin{definition}
	Если множество $D$, на котором определена функция $f$, не ограничено
	сверху (снизу), окрестностью при $x\to+\infty$ ($x\to-\infty$) будем считать:
	\[B'_\varepsilon(+\infty)\coloneq(\varepsilon^{-1}, +\infty)\qquad
	B'_\varepsilon(-\infty)\coloneq(-\infty, \varepsilon^{-1})\]
\end{definition}
\begin{definition}[Бесконечный предел]
	Пусть функция $f$ определена на некотором множестве $D\subset\R$ и
	$a$ -- предельная точка $D$, тогда:
	\[\lim_{x\to a}f(x)=\pm\infty\iff
		\forall\varepsilon>0\ \exists\,\delta>0\colon\forall x\in D\cap
	B'_\delta(a)\ f(x)\in B'_\varepsilon(\pm\infty)\]
\end{definition}
\begin{theorem}
	Определения предела функции по Коши и Гейне эквивалентны.
	\begin{proof}
		Пусть выполняется определение по Коши, то есть:
		\[\forall\varepsilon>0\ \exists\,\delta>0\colon\forall x\in
		D\cap B'_\delta(a)\ \abs{f(x)-A}<\varepsilon\]
		Пусть $x_n\to a,\ x_n\in D\setminus\{a\}$, тогда верно:
		\[\forall\delta>0\ \exists\,N\colon \forall
		n>N\ \abs{x_n-a}<\delta\implies \abs{f(x_n)-A}<\varepsilon\]
		Пусть не выполняется определение по Коши, то есть:
		\[\exists\,\varepsilon>0\colon \forall\delta>0\ \exists\ x\in D\cap
		B'_\delta(a)\colon \abs{f(x)-A}\geqslant\varepsilon\]
		Для последовательности $x_{1/n}\in D\setminus\{a\}$ выполнено
		$x_{1/n}\to a$, но последовательность $f(x_{1/n})$ не сходится к
		$A$, то есть $A$ не является пределом функции $f$ по Гейне.
	\end{proof}
\end{theorem}
\begin{theorem}[Арифметические свойства предела функции]
	Пусть функции $f, g$ определены на некотором множестве $D\subset \R$
	и $a$ -- предельная для $D$ точка, тогда:
	\begin{enumerate}
		\item Единственность предела:
			\[
				\lim_{x\to a}f(x)=A, \lim_{x\to a}f(x)=B\implies A=B
			\]
		\item Линейность предела:
			\[
				\lim_{x\to a}f(x)=A, \lim_{x\to a}g(x)=B \implies
				\forall\alpha,\beta \in\R\ \lim_{x\to a}\left(\alpha f(x)+\beta
				g(x)\right)=\alpha A+\beta B
			\]
		\item Предел произведения:
			\[
				\lim_{x\to a}f(x)=A, \lim_{x\to a}g(x)=B \implies
				\lim_{x\to a}\left(f(x)\cdot g(x)\right)=A\cdot B
			\]
		\item Предел отношения:
			\[
				\lim_{x\to a}f(x)=A, \lim_{x\to a}g(x)=B\neq0, \forall x\in
				D\ g(x)\neq0 \implies
				\lim_{x\to a}\frac{f(x)}{g(x)}=\frac{A}{B}
			\]
	\end{enumerate}
	\begin{proof}
		Доказательства всех арифметических свойств следуют из определения
		предела функции по Гейне.
	\end{proof}
\end{theorem}
\begin{theorem}[Предел функции и неравенства]
	Пусть функции $f, g, h$ определены на некотором множестве $D\subset \R$
	и $a$ -- предельная для $D$ точка, тогда:
	\begin{enumerate}
		\item Переход к пределу в неравенстве:
			\[\exists\,r>0\colon \forall x\in D\cap B'_r(a)\ f(x)\leqslant g(x),
			\lim_{x\to a}f(x)=A, \lim_{x\to a}g(x)=B\implies A\leqslant B\]
		\item Предел зажатой функции:
			\[\exists\,r>0\colon \forall x\in D\cap B'_r(a)\ f(x)\leqslant
				h(x)\leqslant g(x), \lim_{x\to a}f(x)=\lim_{x\to a}g(x)=A\implies
			\lim_{x\to a}h(x)=A\]
		\item Ограниченность функции:
			\[\lim_{x\to a}f(x)=A\implies \exists\,\delta>0, C>0\colon \forall
			x\in D\cap B'_\delta(a)\ \abs{f(x)}\leqslant C\]
		\item Отделимость от нуля:
			\[\lim_{x\to a}f(x)=A\neq0\implies \exists\,\delta>0\colon \forall
			x\in D\cap B'_\delta(a)\ \abs{f(x)}>\frac{\abs{A}}{2}\]
	\end{enumerate}
	\begin{proof}
		Доказательства первых двух свойств следуют из определения предела
		функции по Гейне. Докажем ограниченность функции. По определению
		предела по Коши $\exists\,\delta>0\colon \abs{f(x)-A}<1$, тогда
		выполнено $\abs{f(x)}<1+\abs{A}$. Докажем отделимость от нуля. По
		определению предела по Коши
		$\exists\,\delta>0\colon\abs{f(x)-A}<\abs{A}/2$, тогда
		$\abs{f(x)}>\abs{A}/2$.
	\end{proof}
\end{theorem}
\begin{theorem}[О пределе сложной функции]
	Пусть $f\colon D\to E,\ g\colon E\to\R$, $a$ -- предельная точка
	$D$, $b$ -- предельная точка $E$, $\lim\limits_{x\to
	a}f(x)=b,\ \lim\limits_{y\to b}g(y)=c$, $\exists\,B'_\delta(a)\colon
	\forall x\in B'_\delta(a)\cap D\ f(x)\neq b$. Тогда
	$\lim\limits_{x\to a}g(f(x))=c$.
	\begin{proof}
		Пусть $x_n\to a, x_n\in D, x_n\neq a$. $f(x)\neq b\implies \forall
		x\in B'_\delta(a)\cap D\ \exists\,N\colon\forall n>N\  f(x_n)\neq
		b$. Тогда последовательность $f(x_{N+1}), f(x_{N+2}),\ldots$
		состоит из элементов $E$, причём ни один из этих элементов не
		совпадает с $b$, и эта последовательность сходится к $b$. Тогда
		последовательность $g(f(x_{N+1})), g(f(x_{N+2})),\ldots$ сходится к
		$c$, тогда вся последовательность $\{g(f(x_n))\}$ сходится к $c$.
	\end{proof}
\end{theorem}

\begin{lemma}\label{Предел косинуса}
	\[\lim_{x\to y}\cos x=\cos y\]
	\begin{proof}
		\[\abs{\cos x-\cos
		y}=2\abs{\sin\frac{x+y}{2}\sin\frac{x-y}{2}}\leqslant2\abs{\sin\frac{x-y}{2}}\leqslant\abs{x-y}\]
		Тогда при $\delta=\varepsilon$ выполняется определение предела по Коши.
	\end{proof}
\end{lemma}
\begin{statement}[Первый замечательный предел]
	\[\lim_{x\to0}\frac{\sin x}{x}=1\]
	\begin{proof}
		Сравним при $x\in(0, \pi/2)$ площади сектора единичной окружности с
		площадями двух треугольников:
		\[\frac{1}{2}\cdot 1\cdot 1\cdot \sin x\leqslant\frac{1}{2}\cdot
		1^2\cdot x\leqslant\frac{1}{2}\cdot 1\cdot \tg x\]
		Отсюда при рассмотрении обратных и домножения на положительный
		$\sin x$, имеем:
		\[\cos x\leqslant\frac{\sin x}{x}\leqslant 1\]
		По лемме \ref{Предел косинуса} $\lim\limits_{x\to0}\cos x=1$. Тогда
		по теореме о пределе
		зажатой функции $\lim\limits_{x\to0}\frac{\sin x}{x}=1$.
	\end{proof}
\end{statement}
\begin{statement}[Второй замечательный предел]
	\[\lim_{x\to+\infty}\left(1+\frac{1}{x}\right)^x=e\]
	\begin{proof}
		Введём функции $f$ и $g$:
		\[f(x)=\left(1+\frac{1}{[x]+1}\right)^{[x]}\qquad
		g(x)=\left(1+\frac{1}{[x]}\right)^{[x]+1}\]
		Тогда $f(x)\leqslant\left(1+\frac{1}{x}\right)^x\leqslant g(x)$,
		при этом по определению предела по Гейне:
		\[\lim_{n\to\infty}\left(1+\frac{1}{n+1}\right)^n=\lim_{n\to\infty}\left(1+\frac{1}{n}\right)^{n+1}=e\implies\lim_{x\to+\infty}f(x)=\lim_{x\to+\infty}g(x)=e\]
		Значит утверждение верно из теоремы о пределе зажатой функции.
	\end{proof}
\end{statement}
\begin{theorem}[Критерий Коши]
	Пусть $f\colon D\to \R$ и $a$ -- предельная точка $D$, тогда:
	\[\exists\,\lim_{x\to a}f(x)\iff
		\forall\varepsilon>0\ \exists\,\delta>0\colon \forall x,y\in
	B'_\delta(a)\cap D\ \abs{f(x)-f(y)}<\varepsilon\]
	\begin{proof}
		Если $\lim\limits_{x\to a}f(x)=A$, то найдётся $\delta>0$, при
		котором $\abs{f(x)-A}<\varepsilon/2$, тогда
		$\abs{f(x)-f(y)}\leqslant\abs{f(x)-A}+\abs{A-f(y)}<\varepsilon$.\par

		Если выполнено условие Коши, то для произвольной последовательности
		$x_n\in D\setminus\{a\}, x_n\to a$ последовательность $\{f(x_n)\}$
		фундаментальная, а значит сходится. Пусть $\lim\limits_{n\to
		\infty}f(x_)=A$. Если есть другая аналогичная последовательность
		$y_n$, то рассмотрим такую $z_n$, что $z_{2k-1}=x_k, z_{2k}=y_k$,
		эта последовательность также сходится к $a$ и $\{f(z_n)\}$
		фундаментальна и сходится к $A$. Отсюда $\{f(y_n)\}\to A$, то есть
		предел единственный.
	\end{proof}
\end{theorem}
\begin{definition}
	\[D_a^+\coloneq D\cap(a,+\infty)\qquad D_a^-\coloneq D\cap(-\infty,a)\]
\end{definition}
\begin{definition}
	Пусть $a$ -- предельная точка $D_a^+$ и существует предел $f$ по
	множеству $D_a^+$ в точке $a$. Этот предел называют пределом справа
	функции $f$ в точке $a$ и обозначают $\lim\limits_{x\to a+0}f(x)$
	или $f(a+0)$. Аналогично определяется предел слева
	$\lim\limits_{x\to a-0}f(x)$ или $f(a-0)$ю
\end{definition}
\begin{theorem}[Вейерштрасс]
	Пусть $f$ не убывает и ограничена на множестве $D$, $a$ --
	предельная точка $D_a^-$, тогда существует предел слева:
	\[\lim_{x\to a-0}f(x)=\sup\{f(x)\mid x\in D_a^-\}\]
	Пусть $f$ не убывает и ограничена на множестве $D$, $a$ --
	предельная точка $D_a^+$, тогда существует предел справа:
	\[\lim_{x\to a+0}f(x)=\inf\{f(x)\mid x\in D_a^+\}\]
	Аналогичные утверждения с заменой $\sup$ на $\inf$ и наоборот
	справедливы для не возрастающей функции.
	\begin{proof}
		Пусть $M=\sup\{f(x)\mid x\in D_a^-\}$, тогда
		$\forall\varepsilon>0\ \exists\,x_0\in D_a^-\colon
		M-\varepsilon<f(x_0)$. Поскольку $f$ не убывает на $D_a^-$, то
		$\forall x\in(x_0,a)\cap D_a^-\ M-\varepsilon<f(x_0)\leqslant
		f(x)\leqslant M<M+\varepsilon$. Тогда при $\delta=a-x_0$ получаем,
		что $\forall x\in D_a^-\cap
		B'_\delta(a)\ \abs{f(x)-M}<\varepsilon$. Аналогично для предела
		справа и не возрастающей функции.
	\end{proof}
\end{theorem}

\section{Отношение эквивалентности}

\begin{definition}
	Бинарное отношение $\sim$ на множестве $X$ называется отношением
	эквивалентности, если оно обладает свойствами:
	\begin{center}
		\begin{minipage}{0.65\textwidth}
			\begin{enumerate}[itemsep=0pt]
				\item $\forall x\in X\ x\sim x$ (рефлексивность);
				\item $\forall x,y\in X\ x\sim y\implies y\sim x$ (симметричность);
				\item $\forall x, y, z\in X\ x\sim y, y\sim z\implies x\sim z$
					(транзитивность).
			\end{enumerate}
		\end{minipage}
	\end{center}
\end{definition}
\begin{definition}
	Пусть функции $f$ и $g$ определены и не обращаются в 0 в некоторой
	$B'_\delta(x_0)$. Функции $f$ и $g$ называются эквивалентными при
	$x\to x_0$, если:
	\[\lim_{x\to x_0}\frac{f(x)}{g(x)}=1\]
\end{definition}
\begin{lemma}
	Отношение эквивалентности функций при $x\to x_0$ является отношением
	эквивалетности на множестве функций, определённых и не обращающихся
	в ноль в $B'_\delta(x_0)$.
	\begin{proof}
		\[\tag{Рефлексивность}
		\lim_{x\to x_0}\frac{f(x)}{f(x)}=1\]
		\[\tag{Симметричность}
			\lim_{x\to x_0}\frac{g(x)}{f(x)}=\lim_{x\to
		x_0}\frac{1}{\frac{f(x)}{g(x)}}=\lim_{x\to x_0}\frac{f(x)}{g(x)}=1\]
		\[\tag{Транзитивность}
			\lim_{x\to x_0}\frac{f(x)}{g(x)}=1, \lim_{x\to
			x_0}\frac{g(x)}{h(x)}=1\implies \lim_{x\to
			x_0}\frac{f(x)}{h(x)}=\lim_{x\to
		x_0}\frac{f(x)}{g(x)}\cdot\frac{g(x)}{h(x)}=1\]
	\end{proof}
\end{lemma}
\begin{statement}
	\[x\sim\sin x\sim \tg x\sim \arctg x\sim \arcsin x\sim e^x-1\sim
	\ln(1+x)\sim \sh x\sim \th x\]
\end{statement}
\begin{lemma}
	Пусть функции $f_1(x), f_2(x), g_1(x), g_2(x)$ определены и не
	обращаются в 0 в некоторой $B'_\delta(x_0)$ и $f_1(x)\sim f_2(x),
	g_1(x)\sim g_2(x)$ при $x\to x_0$, тогда при $x\to x_0$:
	\[f_1(x)g_1(x)\sim
	f_2(x)g_2(x)\qquad\frac{f_1(x)}{g_1(x)}\sim\frac{f_2(x)}{g_2(x)}\]
	\begin{proof}
		\[\lim_{x\to x_0}\frac{f_1(x)g_1(x)}{f_2(x)g_2(x)}=\lim_{x\to
		x_0}\frac{f_1(x)}{g_1(x)}\cdot\lim_{x\to x_0}\frac{f_2(x)}{g_2(x)}=1\]
		\[\lim_{x\to x_0}\frac{f_1(x)g_2(x)}{f_2(x)g_1(x)}=\lim_{x\to
		x_0}\frac{f_1(x)}{g_1(x)}\cdot\lim_{x\to x_0}\frac{f_2(x)}{g_2(x)}=1\]
	\end{proof}
\end{lemma}
\begin{lemma}
	Если $f(x)\sim g(x)$ при $x\to x_0$, то:
	\[\lim_{x\to x_0}f(x)=\lim_{x\to x_0}g(x)\]
	При этом если один из пределов не существует, то не существует и другой.
	\begin{proof}
		\[\lim_{x\to x_0}f(x)=\lim_{x\to x_0}\frac{f(x)}{g(x)}\cdot
		g(x)=\lim_{x\to x_0}g(x)\]
	\end{proof}
\end{lemma}
\begin{lemma}
	Пусть $f(y)\sim g(y)$ при $y\to y_0$, $y(x)\to y_0$ при $x\to x_0$ и
	$y(x)\neq y_0$ при $x\in B'_\delta(x_0)$, тогда $f(y(x))\sim
	g(y(x))$ при $x\to x_0$.
	\begin{proof}
		\[\lim_{x\to x_0}\frac{f(y(x))}{g(y(x))}=\lim_{y\to y_0}\frac{f(x)}{g(x)}=1\]
	\end{proof}
\end{lemma}
\begin{definition}[$o$-малое]
	Пусть функции $f,g$ определены в $B'_\delta(x_0)$ и функция $g(x)$
	не обращается в ноль. Функция $f$ называется бесконечно малой
	относительно $g$ при $x\to x_0$, если:
	\[f(x)=o(g(x))\text{ при }x\to x_0\iff\lim_{x\to x_0}\frac{f(x)}{g(x)}=0\]
\end{definition}
\begin{theorem}
	\[f(x)\sim g(x)\text{ при }x\to x_0\iff f(x)-g(x)=o(g(x))\text{ при }x\to x_0\]
	\begin{proof}
		\[\lim_{x\to x_0}\frac{f(x)-g(x)}{g(x)}=0\iff\lim_{x\to
		x_0}\left(\frac{f(x)}{g(x)}-1\right)=0\]
	\end{proof}
\end{theorem}
\begin{definition}[$O$-большое]
	Пусть функции $f,g$ определены в $B'_\delta(x_0)$. Функция $f$
	называется ограниченной относительно функции $g$ при $x\to x_0$, если:
	\[f(x)=O(g(x))\text{ при }x\to x_0\iff \exists\,\delta>0,
	C\in\R\colon\forall x\in B'_\delta(x_0)\ \abs{f(x)}\leqslant C\abs{g(x)}\]
\end{definition}
\begin{theorem}[Свойства $o$-малого и $O$-большого]
	Для функций $f, g$, не обращающихся в 0 в некоторой $B'_\delta(x_0)$
	при $x\to x_0$ справедливо:
	\begin{align*}
		&o(f)\pm o(f)=o(f) & &o(O(f))=o(f) & &o(f)\cdot O(g)=o(fg)\\
		&O(f)\pm O(f)=O(f) & &O(o(f))=o(f) & &O(f)\cdot O(g)=O(fg)\\
		&o(f)=O(f) & &O(O(f))=O(f) &
		&\forall\alpha>0\ \abs{o(f)}^\alpha=o(\abs{f}^\alpha)\\
		& & & & &\forall\alpha>0\ \abs{O(f)}^\alpha=O(\abs{f}^\alpha)
	\end{align*}
\end{theorem}
